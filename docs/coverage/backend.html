package main

import (
        "context"
        "errors"
        "log"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "calculadora-api/internal/api"
)

func main() {
        port := getEnv("PORT", "8080")
        allowedOrigin := getEnv("CORS_ORIGIN", "*")

        server := &http.Server{
                Addr:              ":" + port,
                Handler:           api.NewRouter(allowedOrigin),
                ReadHeaderTimeout: 5 * time.Second,
        }

        // Se escucha Ctrl+C y SIGTERM (lo que manda docker stop) para apagar
        // el servidor sin cortar las peticiones que estén en curso.
        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()

        go func() {
                log.Printf("API escuchando en http://localhost:%s", port)
                if err := server.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
                        log.Fatalf("no se pudo iniciar el servidor: %v", err)
                }
        }()

        <-ctx.Done()
        log.Println("apagando el servidor...")

        shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
        defer cancel()
        if err := server.Shutdown(shutdownCtx); err != nil {
                log.Printf("error al apagar el servidor: %v", err)
        }
}

// getEnv lee una variable de entorno y si no está definida usa el valor por defecto.
func getEnv(key, fallback string) string {
        if value := os.Getenv(key); value != "" {
                return value
        }
        return fallback
}

package api

import (
        "encoding/json"
        "errors"
        "fmt"
        "log"
        "net/http"
)

// Tamaño máximo del body. Para mandar dos números sobra con 1 KB.
const maxBodySize = 1024

// calcRequest es el body que reciben todos los endpoints de operaciones.
// Los campos son punteros para poder diferenciar "no lo mandaron" de "mandaron 0".
type calcRequest struct {
        A *float64 `json:"a"`
        B *float64 `json:"b"`
}

type calcResponse struct {
        Operation string   `json:"operation"`
        A         float64  `json:"a"`
        B         *float64 `json:"b,omitempty"`
        Result    float64  `json:"result"`
}

type errorResponse struct {
        Error string `json:"error"`
}

type binaryFunc func(a, b float64) (float64, error)
type unaryFunc func(a float64) (float64, error)

// binaryHandler arma el handler de una operación que usa dos números (a y b).
func binaryHandler(name string, op binaryFunc) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
                req, ok := readRequest(w, r, true)
                if !ok {
                        return
                }
                result, err := op(*req.A, *req.B)
                respond(w, name, req, result, err)
        }
}

// unaryHandler arma el handler de una operación que usa un solo número (a).
// Si en el body viene b simplemente se ignora.
func unaryHandler(name string, op unaryFunc) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
                req, ok := readRequest(w, r, false)
                if !ok {
                        return
                }
                req.B = nil
                result, err := op(*req.A)
                respond(w, name, req, result, err)
        }
}

// readRequest lee el body y valida que vengan los campos necesarios.
// Si algo está mal ya deja escrito el error en la respuesta y devuelve ok = false.
func readRequest(w http.ResponseWriter, r *http.Request, needsB bool) (calcRequest, bool) {
        var req calcRequest

        r.Body = http.MaxBytesReader(w, r.Body, maxBodySize)
        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
                writeError(w, http.StatusBadRequest, decodeErrorMessage(err))
                return req, false
        }

        if req.A == nil {
                writeError(w, http.StatusBadRequest, "falta el campo 'a'")
                return req, false
        }
        if needsB && req.B == nil {
                writeError(w, http.StatusBadRequest, "falta el campo 'b'")
                return req, false
        }
        return req, true
}

// decodeErrorMessage traduce los errores del decoder de JSON a un mensaje más
// entendible para quien usa la API.
func decodeErrorMessage(err error) string {
        var typeErr *json.UnmarshalTypeError
        var sizeErr *http.MaxBytesError

        switch {
        case errors.As(err, &sizeErr):
                return "el body es demasiado grande"
        case errors.As(err, &typeErr) && typeErr.Field != "":
                return fmt.Sprintf("el campo '%s' debe ser un número válido", typeErr.Field)
        default:
                return `el body debe ser un JSON válido, por ejemplo {"a": 1, "b": 2}`
        }
}

// respond manda el resultado de la operación, o el error si no se pudo calcular.
func respond(w http.ResponseWriter, name string, req calcRequest, result float64, err error) {
        if err != nil {
                // La petición está bien formada pero la operación no se puede hacer
                // (dividir entre cero, raíz de un negativo, etc.), por eso 422 y no 400.
                writeError(w, http.StatusUnprocessableEntity, err.Error())
                return
        }

        writeJSON(w, http.StatusOK, calcResponse{
                Operation: name,
                A:         *req.A,
                B:         req.B,
                Result:    result,
        })
}

func health(w http.ResponseWriter, _ *http.Request) {
        writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
}

func writeJSON(w http.ResponseWriter, status int, data any) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        if err := json.NewEncoder(w).Encode(data); err != nil {
                log.Printf("no se pudo escribir la respuesta: %v", err)
        }
}

func writeError(w http.ResponseWriter, status int, message string) {
        writeJSON(w, status, errorResponse{Error: message})
}

// cors agrega los headers necesarios para que un navegador pueda llamar a la
// API desde otro origen (por ejemplo si el frontend corre sin el proxy).
func cors(allowedOrigin string, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                w.Header().Set("Access-Control-Allow-Origin", allowedOrigin)
                w.Header().Set("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
                w.Header().Set("Access-Control-Allow-Headers", "Content-Type")

                // El preflight del navegador solo necesita los headers de arriba
                if r.Method == http.MethodOptions {
                        w.WriteHeader(http.StatusNoContent)
                        return
                }
                next.ServeHTTP(w, r)
        })
}

// statusRecorder guarda el código de respuesta para poder mostrarlo en el log.
type statusRecorder struct {
        http.ResponseWriter
        status int
}

func (s *statusRecorder) WriteHeader(code int) {
        s.status = code
        s.ResponseWriter.WriteHeader(code)
}

// logRequests imprime una línea por petición: método, ruta, código y duración.
func logRequests(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                start := time.Now()
                rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}

                next.ServeHTTP(rec, r)

                log.Printf("%s %s %d %s", r.Method, r.URL.Path, rec.status, time.Since(start))
        })
}

package api

import (
        "net/http"

        "calculadora-api/internal/calculator"
)

// NewRouter registra todas las rutas de la API.
// allowedOrigin es el origen permitido para CORS ("*" deja pasar a cualquiera).
func NewRouter(allowedOrigin string) http.Handler {
        mux := http.NewServeMux()

        mux.HandleFunc("GET /api/health", health)

        mux.HandleFunc("POST /api/add", binaryHandler("add", calculator.Add))
        mux.HandleFunc("POST /api/subtract", binaryHandler("subtract", calculator.Subtract))
        mux.HandleFunc("POST /api/multiply", binaryHandler("multiply", calculator.Multiply))
        mux.HandleFunc("POST /api/divide", binaryHandler("divide", calculator.Divide))
        mux.HandleFunc("POST /api/power", binaryHandler("power", calculator.Power))
        mux.HandleFunc("POST /api/percentage", binaryHandler("percentage", calculator.Percentage))
        mux.HandleFunc("POST /api/sqrt", unaryHandler("sqrt", calculator.Sqrt))

        return logRequests(cors(allowedOrigin, mux))
}

// Package calculator tiene la lógica matemática de la calculadora.
// No sabe nada de HTTP, así se puede probar y reutilizar por separado.
package calculator

import (
        "errors"
        "math"
)

// Errores que pueden devolver las operaciones. Se exportan para poder
// compararlos con errors.Is desde los handlers y desde los tests.
var (
        ErrDivisionByZero = errors.New("no se puede dividir entre cero")
        ErrNegativeSqrt   = errors.New("no se puede calcular la raíz cuadrada de un número negativo")
        ErrNotRealNumber  = errors.New("el resultado no es un número real")
        ErrOutOfRange     = errors.New("el resultado es demasiado grande")
)

// Add devuelve a + b.
func Add(a, b float64) (float64, error) {
        return checkResult(a + b)
}

// Subtract devuelve a - b.
func Subtract(a, b float64) (float64, error) {
        return checkResult(a - b)
}

// Multiply devuelve a * b.
func Multiply(a, b float64) (float64, error) {
        return checkResult(a * b)
}

// Divide devuelve a / b. Falla si b es cero.
func Divide(a, b float64) (float64, error) {
        if b == 0 {
                return 0, ErrDivisionByZero
        }
        return checkResult(a / b)
}

// Power devuelve a elevado a la b.
func Power(a, b float64) (float64, error) {
        // 0 elevado a un negativo es dividir entre cero (0^-2 = 1/0^2)
        if a == 0 && b < 0 {
                return 0, ErrDivisionByZero
        }
        return checkResult(math.Pow(a, b))
}

// Sqrt devuelve la raíz cuadrada de a. Falla si a es negativo.
func Sqrt(a float64) (float64, error) {
        if a < 0 {
                return 0, ErrNegativeSqrt
        }
        return checkResult(math.Sqrt(a))
}

// Percentage devuelve el a% de b. Por ejemplo Percentage(20, 150) = 30.
func Percentage(a, b float64) (float64, error) {
        return checkResult(a * b / 100)
}

// checkResult revisa que el resultado sea un número finito. Con float64 una
// operación puede terminar en Inf (overflow) o NaN (por ejemplo (-8)^(1/3)
// con math.Pow), y esos valores ni siquiera se pueden mandar en JSON.
func checkResult(result float64) (float64, error) {
        if math.IsNaN(result) {
                return 0, ErrNotRealNumber
        }
        if math.IsInf(result, 0) {
                return 0, ErrOutOfRange
        }
        if result == 0 {
                // evita devolver -0 (pasa con cosas como -5 * 0)
                return 0, nil
        }
        return result, nil
}
